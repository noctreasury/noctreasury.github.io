/* =====================================================
   USER REGISTRATION
   ===================================================== */

function registerUser(data) {

  data = data || {};

  const fullName = clean_(data.fullName);
  const email = normalizeEmail_(data.email);
  const designation = clean_(data.designation);
  const department = clean_(data.department);
  const mobile = normalizeMobile_(data.mobile);
  const recoveryEmail = normalizeEmail_(data.recoveryEmail);
  const password = String(data.password || '');

  if (!fullName) {
    throw new Error('Full Name is required.');
  }

  if (!isValidEmail_(email)) {
    throw new Error('Please enter a valid mailbox email address.');
  }

  if (!designation) {
    throw new Error('Designation is required.');
  }

  if (!department) {
    throw new Error('Department is required.');
  }

  if (!isValidIndianMobile_(mobile)) {
    throw new Error('Please enter a valid 10-digit Indian mobile number.');
  }

  if (recoveryEmail && !isValidEmail_(recoveryEmail)) {
    throw new Error('Please enter a valid Recovery Email address.');
  }

  if (password.length < 8) {
    throw new Error('Password must contain at least 8 characters.');
  }

  if (password !== String(data.confirmPassword || '')) {
    throw new Error('Password and Confirm Password do not match.');
  }

  const sheet = getUsersSheet_();
  const users = getUsersObjects_();

  if (users.some(u => normalizeEmail_(u.Email_Address) === email)) {
    throw new Error('This mailbox email address is already registered.');
  }

  if (
    users.some(
      u => normalizeMobile_(u.Mobile) === mobile
    )
  ) {
    throw new Error('This mobile number is already registered.');
  }

  if (
    recoveryEmail &&
    users.some(
      u => normalizeEmail_(u.Recovery_Email) === recoveryEmail
    )
  ) {
    throw new Error('This Recovery Email is already registered.');
  }

  const userId = generateUserId_(users);

  const now = new Date();

  const row = [
    userId,
    email,
    sha256_(password),
    fullName,
    designation,
    department,
    mobile,
    recoveryEmail,
    'USER',
    'ACTIVE',
    '',
    now,
    '',
    0,
    '',
    'NO',
    'NO'
  ];

  sheet.appendRow(row);

  return {
    ok: true,
    userId: userId,
    email: email,
    recoveryEmail: recoveryEmail,
    message:
      'User registered successfully. Your Mailbox User ID is ' +
      userId +
      '.'
  };
}


/* =====================================================
   SEND RECOVERY OTP
   ===================================================== */

function sendRecoveryOtp(data) {

  data = data || {};

  const email = normalizeEmail_(data.email);
  const recoveryEmail = normalizeEmail_(data.recoveryEmail);

  if (!isValidEmail_(email)) {
    throw new Error('Please enter a valid mailbox email address.');
  }

  if (!isValidEmail_(recoveryEmail)) {
    throw new Error('Please enter a valid Recovery Email address.');
  }

  const user = getUserByEmail_(email);

  if (!user) {
    throw new Error('Mailbox account was not found.');
  }

  if (
    String(user.Status || '').toUpperCase() !== 'ACTIVE'
  ) {
    throw new Error('This mailbox account is not active.');
  }

  const storedRecovery =
    normalizeEmail_(user.Recovery_Email);

  if (!storedRecovery) {
    throw new Error(
      'No Recovery Email is registered for this account. Please use Admin Reset.'
    );
  }

  if (storedRecovery !== recoveryEmail) {
    throw new Error(
      'The Recovery Email does not match our records.'
    );
  }

  const cooldownKey =
    'MAILBOX_OTP_COOLDOWN_' + String(user.User_ID);

  const cache = CacheService.getScriptCache();

  if (cache.get(cooldownKey)) {
    throw new Error(
      'Please wait before requesting another OTP.'
    );
  }

  const otp = generateOtp_();

  const token =
    Utilities.getUuid() +
    Utilities.getUuid().replace(/-/g, '');

  const otpKey =
    'MAILBOX_OTP_' + token;

  const payload = {
    userId: String(user.User_ID),
    email: email,
    recoveryEmail: recoveryEmail,
    otpHash: sha256_(otp),
    created: Date.now(),
    verified: false
  };

  cache.put(
    otpKey,
    JSON.stringify(payload),
    OTP_EXPIRY_SECONDS
  );

  cache.put(
    cooldownKey,
    '1',
    OTP_RESEND_SECONDS
  );

  const subject =
    'Mailbox Password Reset OTP';

  const body =
    'Dear ' +
    String(user.Full_Name || 'User') +
    ',\n\n' +

    'Your OTP for resetting your internal mailbox password is:\n\n' +

    otp +
    '\n\n' +

    'This OTP is valid for 10 minutes.\n\n' +

    'If you did not request a password reset, please contact the NOC/Admin team immediately.\n\n' +

    'Regards,\n' +
    'Internal Mailbox System';

  try {

    MailApp.sendEmail({
      to: recoveryEmail,
      subject: subject,
      body: body,
      name: 'Internal Mailbox'
    });

  } catch (err) {

    cache.remove(otpKey);
    cache.remove(cooldownKey);

    throw new Error(
      'Unable to send OTP email. ' +
      err.message
    );
  }

  return {
    ok: true,
    token: token,
    maskedEmail: maskEmail_(recoveryEmail),
    expiresIn: OTP_EXPIRY_SECONDS
  };
}


/* =====================================================
   VERIFY RECOVERY OTP
   ===================================================== */

function verifyRecoveryOtp(data) {

  data = data || {};

  const token = String(data.token || '').trim();
  const otp = String(data.otp || '').trim();

  if (!token) {
    throw new Error('OTP verification token is missing.');
  }

  if (!/^\d{6}$/.test(otp)) {
    throw new Error('Please enter the 6-digit OTP.');
  }

  const cache = CacheService.getScriptCache();

  const key =
    'MAILBOX_OTP_' + token;

  const raw = cache.get(key);

  if (!raw) {
    throw new Error(
      'OTP has expired or is invalid. Please request a new OTP.'
    );
  }

  const payload = JSON.parse(raw);

  if (
    sha256_(otp).toLowerCase() !==
    String(payload.otpHash).toLowerCase()
  ) {
    throw new Error('Invalid OTP.');
  }

  payload.verified = true;
  payload.verifiedAt = Date.now();

  cache.put(
    key,
    JSON.stringify(payload),
    OTP_EXPIRY_SECONDS
  );

  return {
    ok: true,
    token: token,
    message: 'OTP verified successfully.'
  };
}


/* =====================================================
   RESET PASSWORD USING OTP
   ===================================================== */

function resetPasswordWithOtp(data) {

  data = data || {};

  const token = String(data.token || '').trim();
  const password = String(data.password || '');
  const confirmPassword =
    String(data.confirmPassword || '');

  if (!token) {
    throw new Error('Reset token is missing.');
  }

  if (password.length < 8) {
    throw new Error(
      'New password must contain at least 8 characters.'
    );
  }

  if (password !== confirmPassword) {
    throw new Error(
      'New password and Confirm Password do not match.'
    );
  }

  const cache = CacheService.getScriptCache();

  const key =
    'MAILBOX_OTP_' + token;

  const raw = cache.get(key);

  if (!raw) {
    throw new Error(
      'Password reset session has expired. Please start again.'
    );
  }

  const payload = JSON.parse(raw);

  if (!payload.verified) {
    throw new Error(
      'Please verify the OTP before setting a new password.'
    );
  }

  const user = getUserById_(payload.userId);

  if (!user) {
    throw new Error('User account was not found.');
  }

  updatePassword_(user, password);

  cache.remove(key);

  return {
    ok: true,
    message:
      'Password changed successfully. You can now login to the mailbox.'
  };
}


/* =====================================================
   ADMIN LOGIN
   ===================================================== */

function adminLogin(data) {

  data = data || {};

  const email = normalizeEmail_(data.email);
  const password = String(data.password || '');

  if (!isValidEmail_(email)) {
    throw new Error('Please enter a valid admin email address.');
  }

  if (!password) {
    throw new Error('Admin password is required.');
  }

  const user = getUserByEmail_(email);

  if (!user) {
    throw new Error('Invalid admin credentials.');
  }

  const role =
    String(user.Role || '').toUpperCase();

  if (
    role !== 'ADMIN' &&
    role !== 'SUPERADMIN'
  ) {
    throw new Error(
      'This account does not have administrator access.'
    );
  }

  if (
    String(user.Status || '').toUpperCase() !== 'ACTIVE'
  ) {
    throw new Error('Administrator account is not active.');
  }

  const expected =
    String(user.Password_Hash || '').trim();

  const actual =
    sha256_(password);

  if (
    !expected ||
    expected.toLowerCase() !== actual.toLowerCase()
  ) {
    throw new Error('Invalid admin credentials.');
  }

  const token =
    Utilities.getUuid() +
    Utilities.getUuid().replace(/-/g, '');

  const sessionKey =
    'MAILBOX_ADMIN_SESSION_' + token;

  CacheService.getScriptCache().put(
    sessionKey,
    JSON.stringify({
      userId: String(user.User_ID),
      email: email,
      role: role,
      created: Date.now()
    }),
    3600
  );

  return {
    ok: true,
    token: token,
    admin: {
      userId: user.User_ID,
      email: user.Email_Address,
      name: user.Full_Name,
      role: role
    }
  };
}

/* =====================================================
   CREATE ADMIN RESET REQUEST
   ===================================================== */

function createPasswordResetRequest(data) {

  data = data || {};

  const email = normalizeEmail_(data.email);

  if (!isValidEmail_(email)) {
    throw new Error('Please enter a valid mailbox email address.');
  }

  const user = getUserByEmail_(email);

  if (!user) {
    throw new Error('Mailbox account was not found.');
  }

  if (
    String(user.Status || '').toUpperCase() !== 'ACTIVE'
  ) {
    throw new Error('This mailbox account is not active.');
  }

  const existing =
    getResetRequestObjects_();

  const alreadyPending =
    existing.some(function(r) {

      return (
        String(r.User_ID) ===
          String(user.User_ID) &&

        String(r.Status || '').toUpperCase() ===
          'PENDING'
      );

    });

  if (alreadyPending) {
    throw new Error(
      'A password reset request is already pending for this user.'
    );
  }

  const requestId =
    generateResetRequestId_(existing);

  const sheet =
    getResetRequestsSheet_();

  sheet.appendRow([
    requestId,
    user.User_ID,
    user.Email_Address,
    user.Full_Name,
    new Date(),
    'PENDING',
    '',
    '',
    '',
    '',
    ''
  ]);

  return {
    ok: true,
    requestId: requestId,
    message:
      'Password reset request submitted successfully.'
  };
}


/* =====================================================
   ADMIN GENERATE TEMPORARY PASSWORD
   ===================================================== */

function generateTemporaryPassword(token, requestId) {

  const admin =
    requireAdmin_(token);

  requestId =
    String(requestId || '').trim();

  if (!requestId) {
    throw new Error('Request ID is required.');
  }

  const sheet =
    getResetRequestsSheet_();

  const data =
    sheet.getDataRange().getValues();

  const headers =
    data[0];

  const col = {};

  headers.forEach(function(h, i) {
    col[String(h)] = i;
  });

  let rowNumber = -1;
  let row = null;

  for (let i = 1; i < data.length; i++) {

    if (
      String(data[i][col.Request_ID]) ===
      requestId
    ) {
      rowNumber = i + 1;
      row = data[i];
      break;
    }
  }

  if (!row) {
    throw new Error('Reset request not found.');
  }

  const status =
    String(row[col.Status] || '').toUpperCase();

  if (status !== 'PENDING') {
    throw new Error(
      'This request is no longer pending.'
    );
  }

  const userId =
    String(row[col.User_ID] || '');

  const user =
    getUserById_(userId);

  if (!user) {
    throw new Error('User account was not found.');
  }

  const temporaryPassword =
    generateTemporaryPassword_();

  const passwordHash =
    sha256_(temporaryPassword);

  const now = new Date();

  sheet
    .getRange(rowNumber, col.Status + 1)
    .setValue('APPROVED');

  sheet
    .getRange(rowNumber, col.Approved_By + 1)
    .setValue(admin.email);

  sheet
    .getRange(rowNumber, col.Approved_Date + 1)
    .setValue(now);

  sheet
    .getRange(
      rowNumber,
      col.Temporary_Password_Hash + 1
    )
    .setValue(passwordHash);

  sheet
    .getRange(rowNumber, col.Notes + 1)
    .setValue(
      'Temporary password generated. User must change password after login.'
    );

  updateTemporaryPassword_(user, passwordHash);

  return {
    ok: true,
    requestId: requestId,
    userId: user.User_ID,
    email: user.Email_Address,
    fullName: user.Full_Name,
    temporaryPassword: temporaryPassword,
    message:
      'Temporary password generated. Copy it securely and provide it to the user.'
  };
}


/* =====================================================
   ADMIN REJECT REQUEST
   ===================================================== */

function rejectResetRequest(token, requestId, notes) {

  const admin =
    requireAdmin_(token);

  requestId =
    String(requestId || '').trim();

  const sheet =
    getResetRequestsSheet_();

  const data =
    sheet.getDataRange().getValues();

  const headers =
    data[0];

  const col = {};

  headers.forEach(function(h, i) {
    col[String(h)] = i;
  });

  for (let i = 1; i < data.length; i++) {

    if (
      String(data[i][col.Request_ID]) ===
      requestId
    ) {

      const status =
        String(
          data[i][col.Status] || ''
        ).toUpperCase();

      if (status !== 'PENDING') {
        throw new Error(
          'This request is no longer pending.'
        );
      }

      const rowNumber = i + 1;

      sheet
        .getRange(rowNumber, col.Status + 1)
        .setValue('REJECTED');

      sheet
        .getRange(rowNumber, col.Approved_By + 1)
        .setValue(admin.email);

      sheet
        .getRange(rowNumber, col.Approved_Date + 1)
        .setValue(new Date());

      sheet
        .getRange(rowNumber, col.Notes + 1)
        .setValue(
          clean_(notes) ||
          'Password reset request rejected.'
        );

      return {
        ok: true,
        message: 'Reset request rejected.'
      };
    }
  }

  throw new Error('Reset request not found.');
}

/* =====================================================
   USER LOOKUP
   ===================================================== */

function getUserByEmail_(email) {

  const searchEmail = normalizeEmail_(email);

  if (!searchEmail) {
    return null;
  }

  const sheet = getUsersSheet_();

  const values = sheet.getDataRange().getValues();

  if (values.length < 2) {
    return null;
  }

  const headers = values[0].map(function(header) {
    return String(header || '').trim();
  });

  /*
   * Find Email_Address column.
   */
  const emailCol =
    headers.findIndex(function(header) {
      return header.toLowerCase() === 'email_address';
    });

  if (emailCol === -1) {

    throw new Error(
      'Users sheet configuration error: Email_Address column was not found.'
    );
  }

  /*
   * Build user object using the actual sheet headers.
   */
  for (let i = 1; i < values.length; i++) {

    const row = values[i];

    const rowEmail =
      normalizeEmail_(row[emailCol]);

    if (rowEmail === searchEmail) {

      const user = {};

      headers.forEach(function(header, index) {
        user[header] = row[index];
      });

      user._rowNumber = i + 1;

      return user;
    }
  }

  return null;
}

function getUserById_(userId) {

  userId =
    String(userId || '').trim();

  const users =
    getUsersObjects_();

  for (let i = 0; i < users.length; i++) {

    if (
      String(users[i].User_ID || '').trim() ===
      userId
    ) {
      return users[i];
    }
  }

  return null;
}


/* =====================================================
   UPDATE PASSWORD
   ===================================================== */

function updatePassword_(user, password) {

  const sheet =
    getUsersSheet_();

  const headers =
    getHeaders_(sheet);

  const rowNumber =
    user._rowNumber;

  const passwordCol =
    headers.indexOf('Password_Hash') + 1;

  const failedCol =
    headers.indexOf('Failed_Login_Count') + 1;

  const failedDateCol =
    headers.indexOf('Last_Failed_Login') + 1;

  const sessionCol =
    headers.indexOf('Session_Revoked') + 1;

  const forceCol =
    headers.indexOf('Force_Password_Change') + 1;

  sheet
    .getRange(rowNumber, passwordCol)
    .setValue(sha256_(password));

  if (failedCol > 0) {
    sheet
      .getRange(rowNumber, failedCol)
      .setValue(0);
  }

  if (failedDateCol > 0) {
    sheet
      .getRange(rowNumber, failedDateCol)
      .setValue('');
  }

  if (sessionCol > 0) {
    sheet
      .getRange(rowNumber, sessionCol)
      .setValue('YES');
  }

  if (forceCol > 0) {
    sheet
      .getRange(rowNumber, forceCol)
      .setValue('NO');
  }
}


function updateTemporaryPassword_(user, passwordHash) {

  const sheet =
    getUsersSheet_();

  const headers =
    getHeaders_(sheet);

  const rowNumber =
    user._rowNumber;

  const passwordCol =
    headers.indexOf('Password_Hash') + 1;

  const forceCol =
    headers.indexOf('Force_Password_Change') + 1;

  const sessionCol =
    headers.indexOf('Session_Revoked') + 1;

  sheet
    .getRange(rowNumber, passwordCol)
    .setValue(passwordHash);

  if (forceCol > 0) {
    sheet
      .getRange(rowNumber, forceCol)
      .setValue('YES');
  }

  if (sessionCol > 0) {
    sheet
      .getRange(rowNumber, sessionCol)
      .setValue('YES');
  }
}


/* =====================================================
   ADMIN AUTHENTICATION
   ===================================================== */

function requireAdmin_(token) {

  token =
    String(token || '').trim();

  if (!token) {
    throw new Error('Administrator session is required.');
  }

  const key =
    'MAILBOX_ADMIN_SESSION_' + token;

  const raw =
    CacheService
      .getScriptCache()
      .get(key);

  if (!raw) {
    throw new Error(
      'Administrator session expired. Please login again.'
    );
  }

  return JSON.parse(raw);
}

function getHeaders_(sheet) {

  if (sheet.getLastColumn() === 0) {
    return [];
  }

  return sheet
    .getRange(
      1,
      1,
      1,
      sheet.getLastColumn()
    )
    .getValues()[0]
    .map(String);
}


function setupSheetHeaders_(sheet, headers) {

  if (sheet.getLastRow() === 0) {

    sheet
      .getRange(
        1,
        1,
        1,
        headers.length
      )
      .setValues([headers]);

    return;
  }

  const current =
    getHeaders_(sheet);

  const missing =
    headers.filter(function(header) {
      return current.indexOf(header) === -1;
    });

  if (missing.length) {

    sheet
      .getRange(
        1,
        current.length + 1,
        1,
        missing.length
      )
      .setValues([missing]);

  }
}

function generateUserId_(users) {

  let max = 0;

  users.forEach(function(user) {

    const id =
      String(user.User_ID || '');

    const match =
      id.match(/^USR(\d+)$/i);

    if (match) {

      const number =
        parseInt(match[1], 10);

      if (number > max) {
        max = number;
      }
    }
  });

  return 'USR' +
    String(max + 1).padStart(4, '0');
}


function generateResetRequestId_(requests) {

  let max = 0;

  requests.forEach(function(r) {

    const id =
      String(r.Request_ID || '');

    const match =
      id.match(/^RST(\d+)$/i);

    if (match) {

      const number =
        parseInt(match[1], 10);

      if (number > max) {
        max = number;
      }
    }
  });

  return 'RST' +
    String(max + 1).padStart(5, '0');
}

function generateTemporaryPassword_() {

  const chars =
    'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789@#$%';

  let result = '';

  for (let i = 0; i < 12; i++) {

    result +=
      chars.charAt(
        Math.floor(
          Math.random() * chars.length
        )
      );

  }

  return result;
}


/* =====================================================
   HASHING
   ===================================================== */

function sha256_(text) {

  const bytes =
    Utilities.computeDigest(
      Utilities.DigestAlgorithm.SHA_256,
      String(text),
      Utilities.Charset.UTF_8
    );

  return bytes
    .map(function(b) {

      const v =
        b < 0 ? b + 256 : b;

      const hex =
        v.toString(16);

      return hex.length === 1
        ? '0' + hex
        : hex;

    })
    .join('');
}

function clean_(value) {

  return String(value || '')
    .trim()
    .replace(/\s+/g, ' ');
}


function normalizeEmail_(email) {

  return String(email || '')
    .replace(/\u00A0/g, ' ')
    .replace(/[\u200B-\u200D\uFEFF]/g, '')
    .trim()
    .toLowerCase();
}


function normalizeMobile_(mobile) {

  let value =
    String(mobile || '')
      .replace(/\D/g, '');

  if (value.startsWith('91') && value.length === 12) {
    value = value.substring(2);
  }

  return value;
}


function isValidEmail_(email) {

  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    .test(String(email || ''));
}


function isValidIndianMobile_(mobile) {

  return /^[6-9]\d{9}$/
    .test(String(mobile || ''));
}

   DISPLAY HELPERS
   ===================================================== */

function maskEmail_(email) {

  const parts =
    String(email).split('@');

  if (parts.length !== 2) {
    return '***';
  }

  const name = parts[0];
  const domain = parts[1];

  let maskedName;

  if (name.length <= 2) {
    maskedName = '*'.repeat(name.length);
  } else {
    maskedName =
      name.substring(0, 2) +
      '*'.repeat(
        Math.max(1, name.length - 2)
      );
  }

  return maskedName + '@' + domain;
}


function formatDate_(value) {

  if (!value) {
    return '';
  }

  try {

    const date =
      value instanceof Date
        ? value
        : new Date(value);

    if (isNaN(date.getTime())) {
      return String(value);
    }

    return Utilities.formatDate(
      date,
      Session.getScriptTimeZone(),
      'dd/MM/yyyy HH:mm:ss'
    );

  } catch (e) {

    return String(value);

  }
}

let adminToken = '';
let currentTemporaryPassword = '';


/* =====================================================
   LOGIN
   ===================================================== */

function adminLogin() {

  clearLoginError();

  const email =
    document
      .getElementById('adminEmail')
      .value
      .trim();

  const password =
    document
      .getElementById('adminPassword')
      .value;

  const btn =
    document.getElementById('loginBtn');

  btn.disabled = true;
  btn.textContent = 'Signing in...';

  google.script.run

    .withSuccessHandler(function(result) {

      adminToken = result.token;

      document
        .getElementById('loginPanel')
        .classList.add('hidden');

      document
        .getElementById('dashboard')
        .classList.remove('hidden');

      document
        .getElementById('adminInfo')
        .textContent =
          result.admin.name +
          ' | ' +
          result.admin.email +
          ' | ' +
          result.admin.role;

      btn.disabled = false;
      btn.textContent = 'Login';

      loadRequests();

    })

    .withFailureHandler(function(error) {

      btn.disabled = false;
      btn.textContent = 'Login';

      showLoginError(
        getErrorMessage(error)
      );

    })

    .adminLogin({

      email: email,
      password: password

    });
}


/* =====================================================
   LOAD REQUESTS
   ===================================================== */

function loadRequests() {

  google.script.run

    .withSuccessHandler(function(result) {

      renderRequests(
        result.requests
      );

    })

    .withFailureHandler(function(error) {

      showDashboardMessage(
        getErrorMessage(error),
        false
      );

    })

    .getResetRequests(
      adminToken
    );
}


/* =====================================================
   RENDER TABLE
   ===================================================== */

function renderRequests(requests) {

  const body =
    document.getElementById('requestBody');

  body.innerHTML = '';

  if (!requests || !requests.length) {

    body.innerHTML =
      '<tr>' +
      '<td colspan="7" class="empty">' +
      'No password reset requests found.' +
      '</td>' +
      '</tr>';

    return;
  }

  requests.forEach(function(request) {

    const tr =
      document.createElement('tr');

    const status =
      String(request.status || '')
        .toUpperCase();

    let statusClass =
      'pending';

    if (status === 'APPROVED') {
      statusClass = 'approved';
    }

    if (status === 'COMPLETED') {
      statusClass = 'completed';
    }

    if (status === 'REJECTED') {
      statusClass = 'rejected';
    }

    let action = '';

    if (status === 'PENDING') {

      action =
        '<button class="successBtn" ' +
        'onclick="generateTempPassword(\'' +
        escapeJs(request.requestId) +
        '\')">' +
        'Approve & Generate Password' +
        '</button>' +

        '<button class="danger" ' +
        'style="margin-left:5px;" ' +
        'onclick="rejectRequest(\'' +
        escapeJs(request.requestId) +
        '\')">' +
        'Reject' +
        '</button>';

    } else if (status === 'APPROVED') {

      action =
        '<button class="primary" ' +
        'onclick="completeRequest(\'' +
        escapeJs(request.requestId) +
        '\')">' +
        'Mark Completed' +
        '</button>';

    } else {

      action = '-';

    }

    tr.innerHTML =

      '<td>' +
      escapeHtml(request.requestId) +
      '</td>' +

      '<td>' +
      escapeHtml(request.userId) +
      '</td>' +

      '<td>' +
      escapeHtml(request.fullName) +
      '</td>' +

      '<td>' +
      escapeHtml(request.email) +
      '</td>' +

      '<td>' +
      escapeHtml(request.requestDate) +
      '</td>' +

      '<td>' +
      '<span class="status ' +
      statusClass +
      '">' +
      escapeHtml(status) +
      '</span>' +
      '</td>' +

      '<td>' +
      action +
      '</td>';

    body.appendChild(tr);

  });
}


/* =====================================================
   GENERATE TEMP PASSWORD
   ===================================================== */

function generateTempPassword(requestId) {

  if (
    !confirm(
      'Approve this request and generate a temporary password?'
    )
  ) {
    return;
  }

  google.script.run

    .withSuccessHandler(function(result) {

      currentTemporaryPassword =
        result.temporaryPassword;

      document
        .getElementById('tempPassword')
        .textContent =
          result.temporaryPassword;

      document
        .getElementById('tempPasswordBox')
        .classList.remove('hidden');

      showDashboardMessage(
        'Temporary password generated successfully.',
        true
      );

      loadRequests();

    })

    .withFailureHandler(function(error) {

      showDashboardMessage(
        getErrorMessage(error),
        false
      );

    })

    .generateTemporaryPassword(
      adminToken,
      requestId
    );
}


/* =====================================================
   COPY PASSWORD
   ===================================================== */

function copyTemporaryPassword() {

  if (!currentTemporaryPassword) {
    return;
  }

  navigator.clipboard
    .writeText(
      currentTemporaryPassword
    )
    .then(function() {

      showDashboardMessage(
        'Temporary password copied to clipboard.',
        true
      );

    });

}


/* =====================================================
   HIDE PASSWORD
   ===================================================== */

function hideTemporaryPassword() {

  currentTemporaryPassword = '';

  document
    .getElementById('tempPassword')
    .textContent = '';

  document
    .getElementById('tempPasswordBox')
    .classList.add('hidden');
}


/* =====================================================
   REJECT
   ===================================================== */

function rejectRequest(requestId) {

  const notes =
    prompt(
      'Reason for rejecting this request:',
      ''
    );

  if (notes === null) {
    return;
  }

  google.script.run

    .withSuccessHandler(function(result) {

      showDashboardMessage(
        result.message,
        true
      );

      loadRequests();

    })

    .withFailureHandler(function(error) {

      showDashboardMessage(
        getErrorMessage(error),
        false
      );

    })

    .rejectResetRequest(
      adminToken,
      requestId,
      notes
    );
}


/* =====================================================
   COMPLETE
   ===================================================== */

function completeRequest(requestId) {

  google.script.run

    .withSuccessHandler(function() {

      showDashboardMessage(
        'Request marked as completed.',
        true
      );

      loadRequests();

    })

    .withFailureHandler(function(error) {

      showDashboardMessage(
        getErrorMessage(error),
        false
      );

    })

    .markResetCompleted(
      adminToken,
      requestId
    );
}


/* =====================================================
   LOGOUT
   ===================================================== */

function logout() {

  google.script.run
    .adminLogout(adminToken);

  adminToken = '';

  location.reload();
}


/* =====================================================
   HELPERS
   ===================================================== */

function showLoginError(message) {

  const el =
    document.getElementById('loginError');

  el.textContent = message;
  el.style.display = 'block';
}


function clearLoginError() {

  const el =
    document.getElementById('loginError');

  el.style.display = 'none';
}


function showDashboardMessage(
  message,
  success
) {

  const el =
    document.getElementById(
      'dashboardMessage'
    );

  el.textContent = message;

  el.className =
    'message ' +
    (success ? 'success' : 'error');

  el.style.display = 'block';

  setTimeout(function() {

    el.style.display = 'none';

  }, 5000);
}


function getErrorMessage(error) {

  if (!error) {
    return 'Unknown error.';
  }

  return error.message ||
         String(error);
}


function escapeHtml(value) {

  return String(value || '')
    .replace(/&/g, '&amp;')
    .replace(/</g, '&lt;')
    .replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#039;');
}


function escapeJs(value) {

  return String(value || '')
    .replace(/\\/g, '\\\\')
    .replace(/'/g, "\\'");
}
